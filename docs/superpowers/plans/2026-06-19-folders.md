> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let users create folders (multi-membership collections) and organize saved items into them — assign via a checkbox popover, native desktop drag-and-drop, and per-folder pill removal.

**Architecture:** Folders mirror the existing tags subsystem. A new `folders` table + `item_folders` many-to-many join (item can be in multiple folders). RLS mirrors `item_tags`. A `get_folders_with_counts()` SECURITY INVOKER RPC feeds the sidebar (folders shown even at 0 items, so empty drop targets render). The left-nav **Tags section is replaced by a Folders section**; the top tag filter bar stays. ItemCards become draggable, render folder pills with an ✕-to-remove, and expose an "Add to folder" checkbox popover. Membership writes go through one toggle endpoint with optimistic UI in `ItemsPageClient`.

**Tech Stack:** Next.js 14 App Router, TypeScript, Supabase (Postgres + RLS), shadcn/ui, Tailwind (semantic tokens only), Vitest, native HTML5 Drag-and-Drop API.

-- Folders: user-created multi-membership collections (Jason 2026-06-19)
-- Mirrors tags/item_tags. An item can belong to many folders.

CREATE TABLE public.folders (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid NOT NULL REFERENCES public.users(id) ON DELETE CASCADE,
  name text NOT NULL,
  position int NOT NULL DEFAULT 0,
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX folders_user_id_name_lower_idx ON public.folders(user_id, lower(name));
CREATE INDEX folders_user_id_position_idx ON public.folders(user_id, position, created_at);

CREATE TABLE public.item_folders (
  item_id uuid NOT NULL REFERENCES public.items(id) ON DELETE CASCADE,
  folder_id uuid NOT NULL REFERENCES public.folders(id) ON DELETE CASCADE,
  created_at timestamptz NOT NULL DEFAULT now(),
  PRIMARY KEY (item_id, folder_id)
);
CREATE INDEX item_folders_folder_id_idx ON public.item_folders(folder_id);

-- RLS
ALTER TABLE public.folders ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.item_folders ENABLE ROW LEVEL SECURITY;

CREATE POLICY "folders: select own" ON public.folders FOR SELECT USING (auth.uid() = user_id);
CREATE POLICY "folders: insert own" ON public.folders FOR INSERT WITH CHECK (auth.uid() = user_id);
CREATE POLICY "folders: update own" ON public.folders FOR UPDATE USING (auth.uid() = user_id);
CREATE POLICY "folders: delete own" ON public.folders FOR DELETE USING (auth.uid() = user_id);

-- item_folders: caller must own BOTH the item and the folder
CREATE POLICY "item_folders: select own" ON public.item_folders FOR SELECT
  USING (EXISTS (SELECT 1 FROM public.items i WHERE i.id = item_folders.item_id AND i.user_id = auth.uid()));

CREATE POLICY "item_folders: insert own" ON public.item_folders FOR INSERT
  WITH CHECK (
    EXISTS (SELECT 1 FROM public.items i WHERE i.id = item_folders.item_id AND i.user_id = auth.uid())
    AND EXISTS (SELECT 1 FROM public.folders f WHERE f.id = item_folders.folder_id AND f.user_id = auth.uid())
  );

CREATE POLICY "item_folders: delete own" ON public.item_folders FOR DELETE
  USING (EXISTS (SELECT 1 FROM public.items i WHERE i.id = item_folders.item_id AND i.user_id = auth.uid()));

-- RPC: folders with item counts. SECURITY INVOKER (RLS-enforced, no user param).
-- Unlike get_tags_with_counts, NO `HAVING COUNT > 0` — empty folders must show as drop targets.
CREATE OR REPLACE FUNCTION get_folders_with_counts()
RETURNS TABLE (
  id uuid,
  name text,
  position int,
  item_count int
)
LANGUAGE sql
SECURITY INVOKER
SET search_path = public
AS $$
  SELECT f.id, f.name, f.position,
         COUNT(if2.item_id) FILTER (WHERE i.id IS NOT NULL)::int AS item_count
  FROM folders f
  LEFT JOIN item_folders if2 ON if2.folder_id = f.id
  LEFT JOIN items i ON i.id = if2.item_id AND i.deleted_at IS NULL AND i.archived_at IS NULL
  WHERE f.user_id = auth.uid()
  GROUP BY f.id, f.name, f.position, f.created_at
  ORDER BY f.position ASC, f.created_at ASC;
$$;
```

import { NextRequest, NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'

const MAX_NAME = 40
const MAX_FOLDERS = 100

export async function GET() {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  const { data, error } = await supabase.rpc('get_folders_with_counts')
  if (error) return NextResponse.json({ error: 'Failed to load folders' }, { status: 500 })
  return NextResponse.json({ folders: data ?? [] })
}

export async function POST(request: NextRequest) {
  try {
    const supabase = await createClient()
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

    let body: { name?: string }
    try { body = await request.json() } catch { return NextResponse.json({ error: 'Invalid JSON' }, { status: 400 }) }

    const name = (body.name ?? '').trim()
    if (!name) return NextResponse.json({ error: 'name is required' }, { status: 400 })
    if (name.length > MAX_NAME) return NextResponse.json({ error: `name too long (max ${MAX_NAME})` }, { status: 400 })

    const { data: dup } = await supabase
      .from('folders').select('id').eq('user_id', user.id).ilike('name', name).maybeSingle()
    if (dup) return NextResponse.json({ error: 'A folder with that name already exists' }, { status: 409 })

    const { count } = await supabase
      .from('folders').select('id', { count: 'exact', head: true }).eq('user_id', user.id)
    if ((count ?? 0) >= MAX_FOLDERS) {
      return NextResponse.json({ error: `Folder limit reached (${MAX_FOLDERS})` }, { status: 400 })
    }

    const { data: created, error } = await supabase
      .from('folders').insert({ name, user_id: user.id, position: count ?? 0 })
      .select('id, name, position').single()
    if (error || !created) return NextResponse.json({ error: 'Failed to create folder' }, { status: 500 })

    return NextResponse.json({ folder: created }, { status: 201 })
  } catch (err) {
    console.error('[POST /api/folders]', err instanceof Error ? err.message : err)
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
  }
}
```

```ts
import { NextRequest, NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'

const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i
const MAX_NAME = 40

export async function PATCH(request: NextRequest, { params }: { params: { id: string } }) {
  try {
    const supabase = await createClient()
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    if (!UUID_RE.test(params.id)) return NextResponse.json({ error: 'Invalid id' }, { status: 400 })

    let body: { name?: string; position?: number }
    try { body = await request.json() } catch { return NextResponse.json({ error: 'Invalid JSON' }, { status: 400 }) }

    const update: { name?: string; position?: number } = {}
    if (typeof body.name === 'string') {
      const name = body.name.trim()
      if (!name) return NextResponse.json({ error: 'name cannot be empty' }, { status: 400 })
      if (name.length > MAX_NAME) return NextResponse.json({ error: `name too long (max ${MAX_NAME})` }, { status: 400 })
      // case-insensitive duplicate guard, excluding self
      const { data: dup } = await supabase
        .from('folders').select('id').eq('user_id', user.id).ilike('name', name).neq('id', params.id).maybeSingle()
      if (dup) return NextResponse.json({ error: 'A folder with that name already exists' }, { status: 409 })
      update.name = name
    }
    if (typeof body.position === 'number' && Number.isFinite(body.position)) {
      update.position = Math.trunc(body.position)
    }
    if (Object.keys(update).length === 0) return NextResponse.json({ error: 'Nothing to update' }, { status: 400 })

    const { data, error } = await supabase
      .from('folders').update(update).eq('id', params.id).eq('user_id', user.id)
      .select('id, name, position').maybeSingle()
    if (error) return NextResponse.json({ error: 'Failed to update folder' }, { status: 500 })
    if (!data) return NextResponse.json({ error: 'Not found' }, { status: 404 })
    return NextResponse.json({ folder: data })
  } catch (err) {
    console.error('[PATCH /api/folders/:id]', err instanceof Error ? err.message : err)
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
  }
}

export async function DELETE(_request: NextRequest, { params }: { params: { id: string } }) {
  try {
    const supabase = await createClient()
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    if (!UUID_RE.test(params.id)) return NextResponse.json({ error: 'Invalid id' }, { status: 400 })

    const { error } = await supabase
      .from('folders').delete().eq('id', params.id).eq('user_id', user.id)
    if (error) return NextResponse.json({ error: 'Failed to delete folder' }, { status: 500 })
    return NextResponse.json({ ok: true })
  } catch (err) {
    console.error('[DELETE /api/folders/:id]', err instanceof Error ? err.message : err)
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
  }
}
```

import { NextRequest, NextResponse } from 'next/server'
import { createClient } from '@/lib/supabase/server'

const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i

async function parse(request: NextRequest) {
  try {
    const body = await request.json()
    return typeof body?.folderId === 'string' ? body.folderId : null
  } catch { return null }
}

export async function POST(request: NextRequest, { params }: { params: { id: string } }) {
  try {
    const supabase = await createClient()
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    const folderId = await parse(request)
    if (!UUID_RE.test(params.id) || !folderId || !UUID_RE.test(folderId)) {
      return NextResponse.json({ error: 'Invalid id' }, { status: 400 })
    }
    // RLS insert policy enforces the user owns BOTH item and folder.
    const { error } = await supabase
      .from('item_folders')
      .upsert({ item_id: params.id, folder_id: folderId }, { onConflict: 'item_id,folder_id', ignoreDuplicates: true })
    if (error) return NextResponse.json({ error: 'Failed to add to folder' }, { status: 500 })
    return NextResponse.json({ ok: true })
  } catch (err) {
    console.error('[POST /api/items/:id/folders]', err instanceof Error ? err.message : err)
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
  }
}

export async function DELETE(request: NextRequest, { params }: { params: { id: string } }) {
  try {
    const supabase = await createClient()
    const { data: { user } } = await supabase.auth.getUser()
    if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
    const folderId = await parse(request)
    if (!UUID_RE.test(params.id) || !folderId || !UUID_RE.test(folderId)) {
      return NextResponse.json({ error: 'Invalid id' }, { status: 400 })
    }
    const { error } = await supabase
      .from('item_folders').delete().eq('item_id', params.id).eq('folder_id', folderId)
    if (error) return NextResponse.json({ error: 'Failed to remove from folder' }, { status: 500 })
    return NextResponse.json({ ok: true })
  } catch (err) {
    console.error('[DELETE /api/items/:id/folders]', err instanceof Error ? err.message : err)
    return NextResponse.json({ error: 'Internal server error' }, { status: 500 })
  }
}

    const folder = folders.find((f) => f.id === folderId)

    setOptimisticItems(prev => prev.map(item => {

      const current = item.item_folders ?? []

        body: JSON.stringify({ folderId }),

      if (!res.ok) throw new Error(String(res.status))
    } catch (err) {
      // rollback
      setOptimisticItems(prev => prev.map(item => {
        if (item.id !== itemId) return item
        const current = item.item_folders ?? []
        const reverted = nextInFolder
          ? current.filter(f => f.folders?.id !== folderId)
          : [...current.filter(f => f.folders?.id !== folderId), { folders: { id: folder.id, name: folder.name } }]
        return { ...item, item_folders: reverted }
      }))
      console.error('[folder] toggle failed:', err instanceof Error ? err.message : err)
    }
  }, [folders])
